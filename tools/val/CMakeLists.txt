# Sin codegen validation: compile the translator's emitted shards, registration
# tables and data initializer WITHOUT any graphics/network runtime (no aurora,
# SDL3, Dawn, cryptopp). This proves the generated C++ for all translated
# functions is valid before the full runtime bring-up; the static library is
# never linked here, so unresolved HLE externals are expected at this stage.
#
# Configure (from the repository root):
#   cmake -G Ninja -B build/val -DCMAKE_BUILD_TYPE=Release \
#     -DCMAKE_CXX_COMPILER=".../clang++.exe" tools/val
#   cmake --build build/val
#
# The translator flags mirror runtime/cmake/PublicProducts.cmake +
# RuntimeLibrary.cmake: translated code is compiled with bit-exact FP
# (-fno-fast-math -ffp-contract=off), not the runtime's -ffast-math.

cmake_minimum_required(VERSION 3.25)
project(sin_codegen_validation CXX)

if(NOT CMAKE_CXX_COMPILER_ID MATCHES "^(Clang|AppleClang)$" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Sin validation requires a 64-bit Clang toolchain (LLVM-MinGW)")
endif()
if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Sin validation only supports Release builds")
endif()

set(SIN_WORKSPACE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
get_filename_component(SIN_WORKSPACE_ROOT "${SIN_WORKSPACE_ROOT}" ABSOLUTE)
set(SIN_RUNTIME_DIR "${SIN_WORKSPACE_ROOT}/runtime")
set(SIN_GENERATED "${SIN_WORKSPACE_ROOT}/generated")

include("${SIN_GENERATED}/build_shards/shards.cmake")

if(NOT MKW_BASE_COMMON_SHARDS)
    message(FATAL_ERROR "Translator shard manifest has no shared base shards; run emit-build-shards")
endif()

function(sin_configure_translated target)
    target_include_directories(${target} PRIVATE
        "${SIN_RUNTIME_DIR}/include"
        "${SIN_RUNTIME_DIR}/src"
        # Workspace root, so translator output is spelled "generated/<x>.h".
        "${SIN_WORKSPACE_ROOT}")
    target_compile_definitions(${target} PRIVATE
        TARGET_PC
        MKW_TRANSLATED_GUEST_REGION=${MKW_TRANSLATED_GUEST_REGION})
    set_target_properties(${target} PROPERTIES
        CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON UNITY_BUILD OFF)
    target_compile_options(${target} PRIVATE
        -march=x86-64-v3
        -O2 -fno-fast-math -ffp-contract=off -fno-slp-vectorize
        -mllvm -hot-cold-split=true -w -pipe)
    target_precompile_headers(${target} PRIVATE
        "${SIN_RUNTIME_DIR}/include/guest_abi/guest_abi_pch.h")
endfunction()

add_library(sin_base_shared STATIC ${MKW_BASE_COMMON_SHARDS})
sin_configure_translated(sin_base_shared)

add_library(sin_base_registration OBJECT ${MKW_BASE_REGISTRATION_SOURCES})
sin_configure_translated(sin_base_registration)

# Data initializer + crash symbolization table (generated by generate-data-init).
add_library(sin_data_init OBJECT
    "${SIN_GENERATED}/data_sections_init.cpp"
    "${SIN_GENERATED}/guest_symbol_table.cpp")
sin_configure_translated(sin_data_init)
